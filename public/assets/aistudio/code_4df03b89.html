export function debounceextends (...args: any[]) => any>(
  fn: T,
  delay: number
): DebouncedFunction {
  let timerId: ReturnType<typeof setTimeout> | null = null;
  let lastArgs: Parameters | null = null;
  let result: ReturnType | undefined;

  const debounced = (...args: Parameters) => {
    lastArgs = args;
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(() => {
      result = fn(...(lastArgs as Parameters));
      timerId = null;
      lastArgs = null;
    }, delay);
    return result;
  };

  debounced.cancel = () => {
    if (timerId) clearTimeout(timerId);
    timerId = null;
    lastArgs = null;
  };

  debounced.flush = () => {
    if (timerId && lastArgs) {
      clearTimeout(timerId);
      result = fn(...lastArgs);
      timerId = null;
      lastArgs = null;
    }
    return result;
  };

  return debounced;
}
